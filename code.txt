// ================= FUNGSI VOLUME PROFILE =================

vpBuild(int startIdx, int endIdx, int nRows) =>

    // Volume candle dibagi rata ke semua row yang dilewati; ini estimasi, bukan data tick.
    float lo = 1e10

    float hi = -1e10

    for i = startIdx to endIdx

        int off = bar_index - i

        lo := math.min(lo, low[off])

        hi := math.max(hi, high[off])

    float stp = math.max((hi - lo) / nRows, syminfo.mintick)

    array<float> vols = array.new_float(nRows, 0.0)

    for i = startIdx to endIdx

        int off = bar_index - i

        float v = nz(volume[off])

        if v <= 0

            v := 1.0

        int r0 = math.min(nRows - 1, int((low[off] - lo) / stp))

        int r1 = math.min(nRows - 1, int((high[off] - lo) / stp))

        float share = v / (r1 - r0 + 1)

        for r = r0 to r1

            array.set(vols, r, array.get(vols, r) + share)

    [vols, lo, stp]

vpStats(array<float> vols, float lo, float stp, float pct) =>

    int n = array.size(vols)

    float total = array.sum(vols)

    int pocIdx = array.indexof(vols, array.max(vols))

    float acc = array.get(vols, pocIdx)

    int u = pocIdx

    int d = pocIdx

    while acc < total * pct / 100.0 and (u < n - 1 or d > 0)

        float upV = u < n - 1 ? array.get(vols, u + 1) : -1.0

        float dnV = d > 0 ? array.get(vols, d - 1) : -1.0

        if upV >= dnV

            u += 1

            acc += upV

        else

            d -= 1

            acc += dnV

    [lo + (pocIdx + 0.5) * stp, lo + (u + 1) * stp, lo + d * stp]

    while array.size(vpLb) > 0

    bool sessMode = vpMode == "Session (Day)"

    [vols, pLo, stp] = vpBuild(curStart, bar_index, vpRows)

    [poc, vaH, vaL] = vpStats(vols, pLo, stp, vaPct)

    devPoc := poc

    devVah := vaH

    devVal := vaL

    pPoc := na

    pVah := na

    pVal := na

    if not na(prevStart) and sessStart > prevStart

        [pv, pl, ps] = vpBuild(prevStart, sessStart - 1, vpRows)

        [pp, pvh, pvl] = vpStats(pv, pl, ps, vaPct)

        pPoc := pp

        pVah := pvh

        pVal := pvl

    if showVP

        float maxV = array.max(vols)

        int x0 = sessMode ? sessStart : bar_index + 5

        int xr = bar_index + 25

        for r = 0 to vpRows - 1

            int w = math.max(1, math.round(maxV > 0 ? v / maxV * vpWidth : 0.0))

            float yb = pLo + r * stp

            float yt = yb + stp

            float mid = yb + stp / 2

            bool isPoc = math.abs(mid - poc) < stp * 0.25

            color col = isPoc ? color.new(cPOC, 15) : (mid >= vaL and mid <= vaH ? cVA : cOut)

        array.push(vpLn, line.new(x0, vaH, xr, vaH, xloc=xloc.bar_index, color=cVAL, style=line.style_dashed))

        array.push(vpLn, line.new(x0, vaL, xr, vaL, xloc=xloc.bar_index, color=cVAL, style=line.style_dashed))

        array.push(vpLb, label.new(xr, poc, "POC " + str.tostring(poc, format.mintick), xloc=xloc.bar_index, style=label.style_label_left, color=color.new(cPOC, 80), textcolor=cPOC, size=size.small))

        array.push(vpLb, label.new(xr, vaH, "VAH " + str.tostring(vaH, format.mintick), xloc=xloc.bar_index, style=label.style_label_left, color=color.new(cVAL, 80), textcolor=cVAL, size=size.small))

        array.push(vpLb, label.new(xr, vaL, "VAL " + str.tostring(vaL, format.mintick), xloc=xloc.bar_index, style=label.style_label_left, color=color.new(cVAL, 80), textcolor=cVAL, size=size.small))

        if showPrev and not na(pPoc)

            array.push(vpLn, line.new(prevStart, pPoc, xr, pPoc, xloc=xloc.bar_index, color=cPrev, style=line.style_dotted, width=2))

            array.push(vpLn, line.new(prevStart, pVah, xr, pVah, xloc=xloc.bar_index, color=cPrev, style=line.style_dotted))

            array.push(vpLn, line.new(prevStart, pVal, xr, pVal, xloc=xloc.bar_index, color=cPrev, style=line.style_dotted))

            array.push(vpLb, label.new(xr, pPoc, "yPOC " + str.tostring(pPoc, format.mintick), xloc=xloc.bar_index, style=label.style_label_left, color=color.new(cPrev, 85), textcolor=cPrev, size=size.small))

            array.push(vpLb, label.new(xr, pVah, "yVAH " + str.tostring(pVah, format.mintick), xloc=xloc.bar_index, style=label.style_label_left, color=color.new(cPrev, 85), textcolor=cPrev, size=size.small))

            array.push(vpLb, label.new(xr, pVal, "yVAL " + str.tostring(pVal, format.mintick), xloc=xloc.bar_index, style=label.style_label_left, color=color.new(cPrev, 85), textcolor=cPrev, size=size.small))
